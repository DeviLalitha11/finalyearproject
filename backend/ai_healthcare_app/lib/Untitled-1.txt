class FloatingIcon extends StatelessWidget {
    final IconData icon;
    final Color color;
    final double? top;
    final double? bottom;
    final double? right;
    final double? left;

    const FloatingIcon({
        Key? key,
        required this.icon,
        required this.color,
        this.top,
        this.bottom,
        this.right,
        this.left,
    }) : super(key: key);

    @override
    Widget build(BuildContext context) {
        return Positioned(
            top: top,
            bottom: bottom,
            right: right,
            left: left,
            child: Container(
                padding: const EdgeInsets.all(16),
                decoration: BoxDecoration(
                    shape: BoxShape.circle,
                    color: Colors.white.withOpacity(0.2),
                    border: Border.all(color: Colors.white.withOpacity(0.3), width: 2),
                ),
                child: Icon(icon, color: color, size: 32),
            ),
        );
    }
}
